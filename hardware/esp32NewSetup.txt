// ==== Callbacks для BLE подій сервера ====
class MyServerCallbacks: public BLEServerCallbacks {
    void onConnect(BLEServer* pServer) {
      deviceConnected = true;
      Serial.println("BLE Client Connected");
      // Після підключення клієнта припиняємо рекламу, щоб ESP32 не був видимим для інших пристроїв,
      // коли вже є активне з'єднання. Це економить енергію.
      if (pAdvertising != NULL) {
        pAdvertising->stop();
        Serial.println("BLE Advertising stopped on connect.");
      }
    };

    void onDisconnect(BLEServer* pServer) {
      deviceConnected = false;
      Serial.println("BLE Client Disconnected");
      // Після відключення клієнта, перезапускаємо рекламу, щоб пристрій був знову видимим.
      // Невеликий delay перед стартом реклами може допомогти уникнути проблем.
      if (pAdvertising != NULL) {
        delay(500); // Даємо час Bluetooth стеку для коректного завершення з'єднання
        pAdvertising->start(); // Перезапускаємо рекламу
        Serial.println("Restarting BLE advertising after disconnect.");
      }
    }
};

// ==== Callbacks для характеристик BLE (коли Android записує дані на ESP32) ====
class MyCharacteristicCallbacks: public BLECharacteristicCallbacks {
  void onWrite(BLECharacteristic *pCharacteristic) {
    std::string value = pCharacteristic->getValue();

    if (value.length() > 0) {
      Serial.print("Received value for characteristic ");
      Serial.print(pCharacteristic->getUUID().toString().c_str());
      Serial.print(": '");
      for (int i = 0; i < value.length(); i++)
        Serial.print(value[i]);
      Serial.println("'");

      // Відкриваємо NVS для запису
      preferences.begin("wifi-config", false); // "wifi-config" - назва простору імен, false означає режим читання/запису

      bool success = false; // Змінна для відстеження успіху запису

      if (pCharacteristic->getUUID().equals(SSID_CHARACTERISTIC_UUID)) {
        // Копіюємо отримане значення в буфер, обмежуючи довжину
        strncpy(storedSsid, value.c_str(), sizeof(storedSsid) - 1);
        storedSsid[sizeof(storedSsid) - 1] = '\0'; // Забезпечуємо нульове завершення
        success = preferences.putString("ssid", storedSsid); // Зберігаємо в NVS
        Serial.printf("Saved SSID: '%s' (Success: %s)\n", storedSsid, success ? "true" : "false");
      } else if (pCharacteristic->getUUID().equals(PASS_CHARACTERISTIC_UUID)) {
        strncpy(storedPass, value.c_str(), sizeof(storedPass) - 1);
        storedPass[sizeof(storedPass) - 1] = '\0';
        success = preferences.putString("pass", storedPass);
        Serial.printf("Saved Password: '%s' (Success: %s)\n", storedPass, success ? "true" : "false");
      }
      // ПОВЕРНУТО: Логіка для MQTT Payload (Client ID)
      else if (pCharacteristic->getUUID().equals(MQTT_PAYLOAD_CHARACTERISTIC_UUID)) {
        strncpy(storedMqttPayload, value.c_str(), sizeof(storedMqttPayload) - 1);
        storedMqttPayload[sizeof(storedMqttPayload) - 1] = '\0';
        success = preferences.putString("mqtt_payload", storedMqttPayload);
        Serial.printf("Saved MQTT Payload (Client ID): '%s' (Success: %s)\n", storedMqttPayload, success ? "true" : "false");
      }
      preferences.end(); // Закриваємо NVS

      // Після зміни Wi-Fi або MQTT параметрів, відключаємося від Wi-Fi, щоб ініціювати нове підключення.
      // Це гарантує, що нові налаштування будуть застосовані.
      if (pCharacteristic->getUUID().equals(SSID_CHARACTERISTIC_UUID) ||
          pCharacteristic->getUUID().equals(PASS_CHARACTERISTIC_UUID) ||
          pCharacteristic->getUUID().equals(MQTT_PAYLOAD_CHARACTERISTIC_UUID)) {
        WiFi.disconnect(true);
        Serial.println("Wi-Fi disconnected to apply new settings (or prepare for MQTT re-connection).");
      }
    }
  }
};

// Допоміжна функція для перетворення статусу Wi-Fi в рядок (для читабельності)
const char* wifiStatusToStr(int status) {
  switch(status) {
    case WL_IDLE_STATUS:    return "Idle";
    case WL_NO_SSID_AVAIL:  return "No SSID Available";
    case WL_SCAN_COMPLETED: return "Scan Completed";
    case WL_CONNECTED:      return "Connected";
    case WL_CONNECT_FAILED: return "Connection Failed";
    case WL_CONNECTION_LOST:return "Connection Lost";
    case WL_DISCONNECTED:   return "Disconnected";
    default:                return "Unknown";
  }
}

// ==== Функція для підключення до Wi-Fi ====
void connectToWiFi() {
  // Завантажуємо дані з NVS
  preferences.begin("wifi-config", false);
  String ssidStr = preferences.getString("ssid", "");
  String passStr = preferences.getString("pass", "");
  preferences.end();

  if (ssidStr.length() == 0 || passStr.length() == 0) {
    Serial.println("No Wi-Fi credentials found in NVS. Waiting for BLE configuration.");
    WiFi.disconnect(true); // Переконуємось, що ми відключені
    return;
  }

  // Завжди використовуємо DHCP для отримання IP-адреси
  // Це скидає будь-які попередні налаштування статичної IP.
  Serial.println("Configuring Wi-Fi for DHCP.");
  WiFi.config(INADDR_NONE, INADDR_NONE, INADDR_NONE); // (IP, Gateway, Subnet) - INADDR_NONE означає DHCP

  Serial.printf("Connecting to Wi-Fi: '%s'...\n", ssidStr.c_str());
  WiFi.begin(ssidStr.c_str(), passStr.c_str());

  unsigned long startTime = millis();
  // Чекаємо підключення, але не більше 30 секунд
  while (WiFi.status() != WL_CONNECTED && (millis() - startTime < 30000)) {
    delay(500);
    Serial.print(".");
  }

  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\n✅ Wi-Fi connected!");
    Serial.print("📡 IP Address: ");
    Serial.println(WiFi.localIP());
  } else {
    Serial.printf("\n❌ Failed to connect to Wi-Fi. WiFi Status: %d (%s)\n", WiFi.status(), wifiStatusToStr(WiFi.status()));
    WiFi.disconnect(true); // Відключаємось, якщо підключення не вдалося
  }
}


// ==== Функція для підключення до MQTT ====
void reconnectMqtt() {
  // Завантажуємо Client ID з NVS
  preferences.begin("wifi-config", false);
  String mqttPayloadStr = preferences.getString("mqtt_payload", ""); // Завантажуємо Client ID
  preferences.end();

  // Перевірка, чи Client ID є порожнім
  if (mqttPayloadStr.length() == 0) {
    Serial.println("No MQTT Client ID found in NVS. Cannot connect to MQTT. Please configure via BLE.");
    return;
  }

  // Перевіряємо, чи є з'єднання Wi-Fi, перед тим як намагатися підключитися до MQTT
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("Wi-Fi not connected, cannot connect to MQTT.");
    return;
  }

  client.setServer(MQTT_SERVER, mqttPort); // Використовуємо хардкоджений MQTT_SERVER

  Serial.print("🔄 Connecting to MQTT server: ");
  Serial.print(MQTT_SERVER);
  Serial.print(" as client: ");
  Serial.println(mqttPayloadStr); // Використовуємо завантажений Client ID

  unsigned long startTime = millis();
  // Намагаємося підключитися до MQTT, але не більше 30 секунд
  while (!client.connected() && (millis() - startTime < 30000)) {
    // client.connect() повертає true при успішному підключенні
    // Використовуємо Client ID з NVS, та хардкоджені MQTT_USERNAME, MQTT_AIO_KEY
    if (client.connect(mqttPayloadStr.c_str(), MQTT_USERNAME, MQTT_AIO_KEY)) {
      Serial.println("✅ MQTT connected!");
      // Тут можна підписатися на топіки, якщо потрібно
      // client.subscribe("your/topic");
    } else {
      Serial.printf("❌ MQTT connection failed, rc=%d. Retrying in 5 seconds...\n", client.state());
      delay(5000); // Затримка перед наступною спробою
    }
  }
  if (!client.connected()) {
    Serial.println("❌ Failed to connect to MQTT after multiple attempts.");
  }
}

void setup() {
  Serial.begin(115200);
  Serial.println("Starting ESP32 BLE Configuration & MQTT Client...");

  // Ініціалізація NVS
  if (!preferences.begin("wifi-config", false)) { // "wifi-config" - назва простору імен
    Serial.println("❌ Failed to open NVS namespace 'wifi-config'!");
    return; // Критична помилка, не можемо продовжити без NVS
  }
  Serial.println("NVS namespace 'wifi-config' opened successfully.");

  // Якщо потрібно ОЧИСТИТИ ВСІ ЗБЕРЕЖЕНІ НАЛАШТУВАННЯ (використовувати обережно!):
  // preferences.clear();
  // Serial.println("⚠️ NVS очищено. Будь ласка, перезапишіть дані через BLE.");

  // Завантажуємо збережені дані Wi-Fi та MQTT Client ID при старті
  String loadedSsid = preferences.getString("ssid", "");
  String loadedPass = preferences.getString("pass", "");
  String loadedMqttPayload = preferences.getString("mqtt_payload", ""); // Завантажуємо Client ID

  // Копіюємо String у char[]
  strncpy(storedSsid, loadedSsid.c_str(), sizeof(storedSsid) - 1); storedSsid[sizeof(storedSsid)-1] = '\0';
  strncpy(storedPass, loadedPass.c_str(), sizeof(storedPass) - 1); storedPass[sizeof(storedPass)-1] = '\0';
  strncpy(storedMqttPayload, loadedMqttPayload.c_str(), sizeof(storedMqttPayload) - 1); storedMqttPayload[sizeof(storedMqttPayload)-1] = '\0';

  preferences.end(); // Закриваємо NVS після читання
  Serial.println("NVS read complete.");

  Serial.printf("Loaded from NVS -> SSID: '%s', Pass: '%s', MQTT Client ID: '%s'\n",
                storedSsid, storedPass, storedMqttPayload);
  Serial.printf("Using hardcoded MQTT Server: '%s', User: '%s'\n",
                MQTT_SERVER, MQTT_USERNAME);


  // Налаштування BLE
  BLEDevice::init("ESP32-Config"); // Назва, яку побачить Android-додаток при скануванні
  pServer = BLEDevice::createServer();
  pServer->setCallbacks(new MyServerCallbacks());

  BLEService *pService = pServer->createService(SERVICE_UUID);

  // Характеристика для SSID
  pSsidCharacteristic = pService->createCharacteristic(
                      SSID_CHARACTERISTIC_UUID,
                      BLECharacteristic::PROPERTY_WRITE | BLECharacteristic::PROPERTY_READ
                    );
  pSsidCharacteristic->setCallbacks(new MyCharacteristicCallbacks());
  pSsidCharacteristic->addDescriptor(new BLE2902());

  // Характеристика для пароля
  pPassCharacteristic = pService->createCharacteristic(
                      PASS_CHARACTERISTIC_UUID,
                      BLECharacteristic::PROPERTY_WRITE | BLECharacteristic::PROPERTY_READ
                    );
  pPassCharacteristic->setCallbacks(new MyCharacteristicCallbacks());
  pPassCharacteristic->addDescriptor(new BLE2902());

  // ПОВЕРНУТО: Характеристика для MQTT Payload (client ID / device ID)
  pMqttPayloadCharacteristic = pService->createCharacteristic(
                      MQTT_PAYLOAD_CHARACTERISTIC_UUID,
                      BLECharacteristic::PROPERTY_WRITE | BLECharacteristic::PROPERTY_READ
                    );
  pMqttPayloadCharacteristic->setCallbacks(new MyCharacteristicCallbacks());
  pMqttPayloadCharacteristic->addDescriptor(new BLE2902());

  pService->start(); // Запускаємо BLE сервіс

  // Налаштування реклами BLE
  pAdvertising = pServer->getAdvertising(); // Отримуємо об'єкт реклами з сервера
  pAdvertising->addServiceUUID(SERVICE_UUID); // Додаємо UUID нашого сервісу до рекламного пакета
  pAdvertising->setScanResponse(true); // Дозволяємо сканувати відповідь (більше даних)
  pAdvertising->setMinPreferred(0x06);  // Допомагає Android швидше знайти пристрій
  pAdvertising->setMinPreferred(0x12);
  pAdvertising->start(); // Запуск реклами
  Serial.println("BLE Advertising started. Waiting for connections...");

  // Налаштування кнопок
  pinMode(btnInternal, INPUT_PULLUP); // BOOT кнопка, зазвичай підтягнута до HIGH
  pinMode(btnExternal, INPUT_PULLUP); // Зовнішня кнопка, також підтягнута до HIGH

  // Спроба підключитися до Wi-Fi та MQTT
  connectToWiFi();
  if (WiFi.status() == WL_CONNECTED) { // Підключаємось до MQTT тільки якщо Wi-Fi підключений
    reconnectMqtt();
  }
}

void loop() {
  // Відстежуємо зміну стану підключення BLE для можливих дій
  if (deviceConnected) {
    // Щось робити, коли пристрій підключено по BLE
  }
  if (!deviceConnected && oldDeviceConnected) {
    // Щось робити, коли пристрій BLE відключився (наприклад, перепідключення до Wi-Fi/MQTT)
    oldDeviceConnected = deviceConnected; // Оновлюємо стан
    // Логіка перепідключення Wi-Fi/MQTT вже ініціюється в onWrite або окремо в loop()
  }
  if (deviceConnected && !oldDeviceConnected) {
    // Щось робити, коли пристрій BLE підключився
    oldDeviceConnected = deviceConnected; // Оновлюємо стан
  }

  // Автоматичне перепідключення до Wi-Fi, якщо від'єдналися
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("Wi-Fi disconnected or not connected. Attempting to connect...");
    connectToWiFi(); // Ця функція перевірить наявність облікових даних в NVS
    delay(5000); // Чекаємо 5 секунд перед наступною спробою Wi-Fi підключення
  }


  // Автоматичне перепідключення до MQTT, якщо Wi-Fi підключений, але MQTT ні
  if (WiFi.status() == WL_CONNECTED && !client.connected()) {
    Serial.println("MQTT disconnected. Attempting to reconnect...");
    reconnectMqtt();
    delay(5000); // Чекаємо 5 секунд перед наступною спробою MQTT підключення
  }

  // Обов'язково викликати client.loop() для підтримки MQTT з'єднання та обробки вхідних повідомлень
  if (client.connected()) {
    client.loop();
  }

  // ---------- Обробка BOOT кнопки ----------
  int readingInt = digitalRead(btnInternal);
  if (readingInt == LOW && lastInternal == HIGH) { // Перехід з HIGH на LOW (натискання)
    delay(20);  // Дебаунс
    if (digitalRead(btnInternal) == LOW) { // Перевіряємо, чи кнопка все ще натиснута
      Serial.println("Кнопка BOOT натиснута");
      if (client.connected()) {
        client.publish(MQTT_TOPIC_BUTTON, storedMqttPayload); // Публікуємо збережений Client ID
        Serial.printf("Published '%s' to '%s' topic.\n", storedMqttPayload, MQTT_TOPIC_BUTTON);
      } else {
        Serial.println("MQTT Not connected, cannot publish button event.");
      }
    }
  }
  lastInternal = readingInt; // Оновлюємо останній стан кнопки

  // ---------- Обробка Зовнішньої кнопки ----------
  int readingExt = digitalRead(btnExternal);
  if (readingExt == LOW && lastExternal == HIGH) { // Перехід з HIGH на LOW (натискання)
    delay(20);  // Дебаунс
    if (digitalRead(btnExternal) == LOW) { // Перевіряємо, чи кнопка все ще натиснута
      Serial.println("Зовнішня кнопка натиснута");
      if (client.connected()) {
        client.publish(MQTT_TOPIC_BUTTON, storedMqttPayload); // Публікуємо збережений Client ID
        Serial.printf("Published '%s' to '%s' topic.\n", storedMqttPayload, MQTT_TOPIC_BUTTON);
      } else {
        Serial.println("MQTT Not connected, cannot publish button event.");
      }
    }
  }
  lastExternal = readingExt; // Оновлюємо останній стан кнопки

  // Короткий delay, щоб не завантажувати процесор повністю, якщо немає інших активних операцій
  delay(100);
}
